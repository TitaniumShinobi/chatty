// Express-based backend handler for ZIP file upload and auto-persistence to Supabase


const express = require('express');
const multer = require('multer');
const AdmZip = require('adm-zip');
const { createClient } = require('@supabase/supabase-js');
const path = require('path');
const crypto = require('crypto');
const fs = require('fs');


const router = express.Router();


const supabase = createClient(process.env.SUPABASE_URL, process.env.SUPABASE_SERVICE_ROLE_KEY);
const upload = multer({ dest: 'tmp_uploads/' });


// Helper to create hash
const sha256 = (buffer) => crypto.createHash('sha256').update(buffer).digest('hex');


router.post('/api/upload-zip', upload.single('file'), async (req, res) => {
  try {
    if (!req.file || path.extname(req.file.originalname).toLowerCase() !== '.zip') {
      return res.status(400).json({ error: 'Only .zip files are supported.' });
    }


    const zipPath = req.file.path;
    const zip = new AdmZip(zipPath);
    const zipEntries = zip.getEntries();


    const gptId = req.body.gptId || 'unknown-gpt';
    const logs = [];


    for (const entry of zipEntries) {
      if (entry.isDirectory || entry.entryName.startsWith('__MACOSX')) continue;
      const fileBuffer = entry.getData();


      const ext = path.extname(entry.entryName).toLowerCase();
      const fileName = path.basename(entry.entryName);
      const subpath = entry.entryName.replace(/\\/g, '/');


      // Skip unsupported file types
      if (!['.pdf', '.txt', '.md', '.json'].includes(ext)) continue;


      const hash = sha256(fileBuffer);
      const pathInStorage = `gpts/${gptId}/${subpath}`;


      const { error } = await supabase.storage
        .from('vault_files')
        .upload(pathInStorage, fileBuffer, {
          contentType: mimeType(ext),
          upsert: true
        });


      logs.push({ file: subpath, size: fileBuffer.length, hash, error });
    }


    fs.unlinkSync(zipPath); // Cleanup temp ZIP
    res.json({ uploaded: logs });
  } catch (err) {
    console.error('[ZIP UPLOAD ERROR]', err);
    res.status(500).json({ error: 'Failed to process zip file.' });
  }
});


function mimeType(ext) {
  return {
    '.txt': 'text/plain',
    '.md': 'text/markdown',
    '.json': 'application/json',
    '.pdf': 'application/pdf',
  }[ext] || 'application/octet-stream';
}


module.exports = router;
