✅ handleZipUpload() — Chatty Frontend Patch (React or JS)

Replace the broken file-by-file upload logic with this:

async function handleZipUpload(zipFile) {
  const formData = new FormData();
  formData.append("file", zipFile);

  try {
    const response = await fetch("/api/vault/knowledge-files/upload", {
      method: "POST",
      body: formData,
      credentials: "include" // crucial if your backend checks auth
    });

    if (!response.ok) {
      const errorText = await response.text();
      throw new Error(`Upload failed: ${response.status} ${errorText}`);
    }

    const result = await response.json();

    console.log("Upload complete:", result);

    // Optional: visual feedback in Chatty
    alert(`Uploaded ${result.created.length} new files and updated ${result.updated.length} files!`);

  } catch (err) {
    console.error("Zip upload error:", err);
    alert("Something went wrong uploading the ZIP: " + err.message);
  }
}
