function formatMarkdownTranscript(messages) {
  return messages.map(msg => {
    const roleLabel = msg.role === 'user' ? '**User**' : '**Assistant**';

    const hasAttachments = Array.isArray(msg.attachments) && msg.attachments.length > 0;
    const hasContent = typeof msg.content === 'string' && msg.content.trim() !== '';

    let attachmentBlock = '';
    if (hasAttachments) {
      attachmentBlock =
        '📎 attached:\n' +
        msg.attachments
          .map(att => `- ${att.filename || att.name || 'file'}`)
          .join('\n') +
        '\n\n';
    }

    // 🔒 Canonical placeholder rule
    const content =
      hasContent
        ? msg.content
        : hasAttachments
          ? '[attached]'
          : '';

    return `${roleLabel}:\n${attachmentBlock}${content}`.trim();
  }).join('\n\n');
}
